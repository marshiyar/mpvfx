cmake_minimum_required(VERSION 3.20)
project(mpvfx_library LANGUAGES CXX)
if(APPLE)
  execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE MACOS_SDK OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  find_library(SYSTEM_SQLITE sqlite3 PATHS "${MACOS_SDK}/usr/lib" NO_DEFAULT_PATH REQUIRED)
else()
  find_package(SQLite3 REQUIRED)
endif()
if(NOT EXISTS "${NODE_API_INCLUDE_DIR}/node_api.h")
  message(FATAL_ERROR "node_api.h not found; set NODE_API_INCLUDE_DIR")
endif()
add_library(mpvfx_library MODULE library_node.cpp)
target_compile_features(mpvfx_library PRIVATE cxx_std_17)
target_include_directories(mpvfx_library PRIVATE "${NODE_API_INCLUDE_DIR}")
if(APPLE)
  target_link_libraries(mpvfx_library PRIVATE "${SYSTEM_SQLITE}")
else()
  target_link_libraries(mpvfx_library PRIVATE SQLite::SQLite3)
endif()
if(WIN32)
  if(NOT EXISTS "${NODE_API_LIBRARY}")
    message(FATAL_ERROR "Electron node.lib not found; set NODE_API_LIBRARY")
  endif()
  if(NOT EXISTS "${NODE_API_DELAY_LOAD_HOOK}")
    message(FATAL_ERROR "Electron delay-load hook not found; set NODE_API_DELAY_LOAD_HOOK")
  endif()
  target_sources(mpvfx_library PRIVATE "${NODE_API_DELAY_LOAD_HOOK}")
  target_compile_definitions(mpvfx_library PRIVATE HOST_BINARY="node.exe")
  target_link_libraries(mpvfx_library PRIVATE "${NODE_API_LIBRARY}")
  target_link_libraries(mpvfx_library PRIVATE delayimp.lib)
  target_link_options(mpvfx_library PRIVATE /DELAYLOAD:node.exe)
  # Visual Studio otherwise appends /Release and the package misses the addon.
  set_target_properties(mpvfx_library PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}"
    RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}")
endif()
if(MSVC)
  target_compile_options(mpvfx_library PRIVATE /W4 /WX)
else()
  target_compile_options(mpvfx_library PRIVATE -Wall -Wextra -Werror)
endif()
set_target_properties(mpvfx_library PROPERTIES PREFIX "" SUFFIX ".node")
if(APPLE)
  target_link_options(mpvfx_library PRIVATE -undefined dynamic_lookup)
endif()
